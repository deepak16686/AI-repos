## Example 1 - CLAUDE.md: unreasonably effective, but it bloats.

Talk: a static markdown file injected at session start steers the agent very well,
but every rule you add is paid for in EVERY session. Shows the token cost of a

```python
"""Example 1 - CLAUDE.md: unreasonably effective, but it bloats.

Talk: a static markdown file injected at session start steers the agent very well,
but every rule you add is paid for in EVERY session. Shows the token cost of a
growing CLAUDE.md versus a slim one that points to on-demand memory.
"""
from memlib import tokens

RULE = "- Rule {i}: use the approved base image, registry host and stage order for project type {i}.\n"


def claude_md(n_rules):
    return "# CLAUDE.md\n" + "".join(RULE.format(i=i) for i in range(n_rules))


SESSIONS_PER_DAY = 100
print(f"{'rules':>6} {'tokens/session':>15} {'tokens/day (100 sessions)':>27}")
for n in (10, 50, 200, 1000):
    t = tokens(claude_md(n))
    print(f"{n:>6} {t:>15,} {t * SESSIONS_PER_DAY:>27,}")

slim = claude_md(10) + "- For anything else: grep memory/index.md, then read only the file you need.\n"
fat = claude_md(1000)
print()
print(f"slim CLAUDE.md (10 rules + pointer) : {tokens(slim):,} tokens/session")
print(f"fat  CLAUDE.md (1000 rules)         : {tokens(fat):,} tokens/session")
print(f"saving per session                  : {tokens(fat) - tokens(slim):,} tokens "
      f"({100 * (1 - tokens(slim) / tokens(fat)):.0f}%)")
```

## Example 2 - Memory tools: the agent reads and writes memory in-band.

Talk: let the agent decide, DURING the session, when to look things up and when to
write a lesson down. Session 1 hits a Maven failure and records the fix; session 2
greps memory first and succeeds on the first attempt (the "continual learning" feel).
The agent here is a deterministic stand-in for an LLM; the memory behaviour is real.


<details><summary>Source: <code>ex02_memory_tool_loop.py</code></summary>

```python
"""Example 2 - Memory tools: the agent reads and writes memory in-band.

Talk: let the agent decide, DURING the session, when to look things up and when to
write a lesson down. Session 1 hits a Maven failure and records the fix; session 2
greps memory first and succeeds on the first attempt (the "continual learning" feel).
The agent here is a deterministic stand-in for an LLM; the memory behaviour is real.
"""
import tempfile
from memlib import MemoryStore

FIX = "nexus-mirror"
ME = "agent:pipeline-agent"
LESSONS = "agents/pipeline-agent/lessons.md"


def run_pipeline(settings):
    if FIX in settings:
        return True, "BUILD SUCCESS"
    return False, "ERROR: mvn dependency resolution failed: repo.maven.apache.org unreachable"


def agent_session(store, session):
    print(f"[{session}] task: generate + run pipeline for a Java repo")
    settings = set()
    hits = store.grep("maven", prefix="agents/pipeline-agent")          # in-band READ
    if hits:
        print(f"[{session}] memory hit: {hits[0][0]}:{hits[0][1]}  {hits[0][2]!r}")
        settings.add(FIX)
    else:
        print(f"[{session}] memory hit: none")
    attempts = 0
    while True:
        attempts += 1
        ok, msg = run_pipeline(settings)
        print(f"[{session}]   attempt {attempts}: {msg}")
        if ok:
            break
        settings.add(FIX)                                                # reason from error
        text, h = store.read(LESSONS, ME, missing_ok=True)               # in-band WRITE
        text += "- maven: builds must use the nexus-mirror setting (central is unreachable)\n"
        store.write(LESSONS, text, ME, session, expected_hash=h, reason="learned from failed build")
        print(f"[{session}]   wrote lesson to {LESSONS}")
    print(f"[{session}] done in {attempts} attempt(s)\n")
    return attempts


store = MemoryStore(tempfile.mkdtemp())
a1 = agent_session(store, "session-1")
a2 = agent_session(store, "session-2")
print(f"attempts: session-1={a1}, session-2={a2}  ->  improvement without any retraining")
```

## Example 3 - Skills: progressive disclosure.

Talk: the agent only sees each skill's front matter (title + description). It loads
the full body only when the task matches - like scanning book titles on a shelf and
pulling off the French dictionary when someone speaks French.


<details><summary>Source: <code>ex03_skills_progressive_disclosure.py</code></summary>

```python
"""Example 3 - Skills: progressive disclosure.

Talk: the agent only sees each skill's front matter (title + description). It loads
the full body only when the task matches - like scanning book titles on a shelf and
pulling off the French dictionary when someone speaks French.
"""
import re
import tempfile
from pathlib import Path
from memlib import tokens

SKILLS = {
    "gitlab-pipeline": ("Generate an 8-stage GitLab CI pipeline: compile, build, test, sast, quality, security, push, notify",
                        "pipeline gitlab ci stages"),
    "dockerfile": ("Create a multi-stage Dockerfile after analysing the repository",
                   "dockerfile docker image multi-stage"),
    "incident-triage": ("Diagnose a failed pipeline and propose a fix",
                        "failed failure triage diagnose"),
}

root = Path(tempfile.mkdtemp())
for name, (desc, _) in SKILLS.items():
    body = "".join(f"{i}. Detailed procedural step {i} for {name}: check inputs, apply golden rules, validate output.\n"
                   for i in range(1, 61))
    (root / f"{name}.md").write_text(f"---\nname: {name}\ndescription: {desc}\n---\n{body}")


def front_matter(path):
    m = re.match(r"---\n(.*?)\n---\n", path.read_text(), re.S)
    return dict(line.split(": ", 1) for line in m.group(1).splitlines())


index = {p.stem: front_matter(p) for p in root.glob("*.md")}
all_bodies = sum(tokens(p.read_text()) for p in root.glob("*.md"))
titles = sum(tokens(f"{v['name']}: {v['description']}") for v in index.values())

print(f"skills on shelf: {len(index)}")
print(f"if ALL skills were loaded into context : {all_bodies:,} tokens")
print(f"progressive disclosure, titles only    : {titles:,} tokens\n")

for request in ("please generate a gitlab ci pipeline for this java repo",
                "write a multi-stage dockerfile for the service",
                "tell me a joke"):
    words = set(request.lower().split())
    scored = sorted(((len(words & set((v["name"] + " " + v["description"]).lower().replace("-", " ").split())), n)
                     for n, v in index.items()), reverse=True)
    score, best = scored[0]
    if score < 2:
        print(f"request: {request!r}\n  -> no skill matched, context cost = {titles:,} tokens")
        continue
    body = (root / f"{best}.md").read_text()
    print(f"request: {request!r}\n  -> matched skill '{best}' (score {score}), "
          f"loaded body: context cost = {titles + tokens(body):,} tokens "
          f"(vs {all_bodies:,} if everything were preloaded)")
```

## Example 6 - Concurrency with a hash check (production principle #2).

Talk: before editing, take a hash; draft the edit; take another hash right before
writing; if they differ somebody else changed the file -> re-read and retry.
Eight agents append a note to the same file at the same moment, without and with
the check.


<details><summary>Source: <code>ex06_concurrency_hash_check.py</code></summary>

```python
"""Example 6 - Concurrency with a hash check (production principle #2).

Talk: before editing, take a hash; draft the edit; take another hash right before
writing; if they differ somebody else changed the file -> re-read and retry.
Eight agents append a note to the same file at the same moment, without and with
the check.
"""
import tempfile
import threading
from memlib import MemoryStore, ConflictError

N = 8
PATH = "shared/notes.md"


def run(use_check):
    store = MemoryStore(tempfile.mkdtemp())
    barrier = threading.Barrier(N)
    retries = [0] * N

    def worker(i):
        actor = f"agent:a{i}"
        text, h = store.read(PATH, actor, missing_ok=True)
        barrier.wait()                       # all agents have read the SAME version
        while True:
            try:
                store.write(PATH, text + f"- note from a{i}\n", actor, f"s{i}",
                            expected_hash=h if use_check else None)
                return
            except ConflictError:            # somebody wrote first: re-read, re-draft, retry
                retries[i] += 1
                text, h = store.read(PATH, actor)

    threads = [threading.Thread(target=worker, args=(i,)) for i in range(N)]
    [t.start() for t in threads]
    [t.join() for t in threads]
    lines = len(store.read(PATH)[0].splitlines())
    return lines, sum(retries)


for label, check in (("WITHOUT hash check (last writer wins)", False), ("WITH hash check + retry", True)):
    lines, retries = run(check)
    print(f"{label}")
    print(f"  agents: {N}   notes in final file: {lines}   lost updates: {N - lines}   conflicts detected and retried: {'>= ' + str(N - 1) if retries >= N - 1 else retries}")
```

## Example 8 - Portability (production principle #4).

JSON bundle, imported into a completely separate store, and also read by a "second

```python
"""Example 8 - Portability (production principle #4).
```

## Example 9 - Stale and maliciously injected memory.

Talk: memories go stale, can be written wrongly, or be injected by someone trying to
prompt-inject your agents. Guardrails: provenance, a screening step before the
review gate, and freshness checks. Deterministic rule-based screening (a real system
could add an LLM classifier on top).


<details><summary>Source: <code>ex09_stale_and_injected_memory.py</code></summary>

```python
"""Example 9 - Stale and maliciously injected memory.

Talk: memories go stale, can be written wrongly, or be injected by someone trying to
prompt-inject your agents. Guardrails: provenance, a screening step before the
review gate, and freshness checks. Deterministic rule-based screening (a real system
could add an LLM classifier on top).
"""
import re
import tempfile
from datetime import date
from memlib import MemoryStore

TODAY = date(2026, 10, 16)
MAX_AGE_DAYS = 90

INJECTION = [
    (r"ignore (all )?(previous|prior) instructions", "instruction override"),
    (r"curl[^\n]*\|\s*(sh|bash)", "remote code execution"),
    (r"(skip|disable|bypass)[^\n]*(trivy|sonar|security|quality gate)", "weakens security gate"),
    (r"(send|post|upload)[^\n]*(token|password|secret|credential)", "credential exfiltration"),
]


def screen(text):
    return [why for rx, why in INJECTION if re.search(rx, text, re.I)]


store = MemoryStore(tempfile.mkdtemp())
```

## Example 10 - "Dreaming": offline consolidation of memory (illustrative).

NOTE: the talk was cut off before this section, so this is a reconstruction of the
idea, not a copy of Anthropic's implementation. Between sessions a background pass
reads transcripts, finds lessons that recur, drops stale/contradicted memory, and
PROPOSES an updated memory file that goes through the review gate.


<details><summary>Source: <code>ex10_dreaming_consolidation.py</code></summary>

```python
"""Example 10 - "Dreaming": offline consolidation of memory (illustrative).

NOTE: the talk was cut off before this section, so this is a reconstruction of the
idea, not a copy of Anthropic's implementation. Between sessions a background pass
reads transcripts, finds lessons that recur, drops stale/contradicted memory, and
PROPOSES an updated memory file that goes through the review gate.
"""
import re
import tempfile
from collections import defaultdict
from datetime import date
from memlib import MemoryStore

TODAY = date(2026, 10, 16)
store = MemoryStore(tempfile.mkdtemp())
store.write("project/lessons.md",
            "- [topic=registry] [verified 2026-02-01] Push images to ai-nexus:5000\n"
            "- [topic=sonar] [verified 2026-09-10] SonarQube gate must pass before push\n"
            "- [topic=old-runner] [verified 2026-01-05] Use runner tag legacy-docker\n",
            "human:alice", "curation", reason="seed")

# transcripts: (session, kind, topic, text)
events = [
    ("s1", "error", "registry", "push denied on ai-nexus:5000"), ("s1", "fix", "registry", "Push images to ai-nexus:5001"),
    ("s2", "error", "registry", "push denied on ai-nexus:5000"), ("s2", "fix", "registry", "Push images to ai-nexus:5001"),
    ("s3", "error", "maven", "central unreachable"),            ("s3", "fix", "maven", "Use nexus-mirror setting for Maven"),
    ("s4", "error", "maven", "central unreachable"),            ("s4", "fix", "maven", "Use nexus-mirror setting for Maven"),
    ("s5", "error", "trivy", "scan timeout"),                   ("s5", "fix", "trivy", "Raise trivy timeout to 10m"),
    ("s6", "info", "sonar", "gate passed as expected"),
]
print(f"input: {len({e[0] for e in events})} session transcripts, {len(events)} events\n")

seen = defaultdict(lambda: defaultdict(set))                 # topic -> fix text -> sessions
for sess, kind, topic, text in events:
    if kind == "fix":
        seen[topic][text].add(sess)

existing = {}
for line in store.read("project/lessons.md")[0].splitlines():
    m = re.match(r"- \[topic=(\S+)\] \[verified (\S+)\] (.*)", line)
    existing[m.group(1)] = (date.fromisoformat(m.group(2)), m.group(3))

new_lines, report = [], []
for topic, (d, text) in existing.items():
    if topic in seen and text not in seen[topic]:
        fix = max(seen[topic], key=lambda t: len(seen[topic][t]))
        report.append(f"REPLACE  topic={topic}: contradicted by {len(seen[topic][fix])} sessions")
        new_lines.append(f"- [topic={topic}] [verified {TODAY}] {fix}")
    elif (TODAY - d).days > 90 and topic not in seen:
        report.append(f"PRUNE    topic={topic}: unverified for {(TODAY - d).days} days")
    else:
        new_lines.append(f"- [topic={topic}] [verified {d}] {text}")
for topic, fixes in seen.items():
    for text, sessions in fixes.items():
        if topic in existing:
            continue
        if len(sessions) >= 2:
            report.append(f"PROMOTE  topic={topic}: seen in {len(sessions)} sessions")
            new_lines.append(f"- [topic={topic}] [verified {TODAY}] {text}")
        else:
            report.append(f"WATCH    topic={topic}: only 1 session, not promoted yet")

print("dreaming report:")
print("\n".join("  " + r for r in report))
proposal = "\n".join(new_lines) + "\n"
p = store.propose("agent:dreamer", "project/lessons.md", proposal, "dream-2026-10-16", reason="nightly consolidation")
print(f"\nproposal written to {p}; waiting at the review gate ...")
store.approve("human:alice", p, "review-2026-10-16")

print("\nmemory before (v1):\n" + store.versions("project/lessons.md")[0]["content"].rstrip())
print("\nmemory after (v2, approved by human):\n" + store.read("project/lessons.md")[0].rstrip())
last = store.versions("project/lessons.md")[-1]
print(f"\nprovenance: v{last['v']} by {last['actor']} - {last['reason']}")
```

## Example 11 - Swapping the model: context is the durable asset.

Talk: a newer, smarter model still knows nothing about YOUR environment, because
that knowledge is orthogonal to model intelligence. If your context lives outside the
model (memory files, skills, CLAUDE.md), a new model plugs into it and is instantly
useful; without it, it re-learns every local fact the hard way.

Toy model: 4 local facts must be right (registry, maven, sonar, runner) plus 1
generic hard step. The old model (v1) is weaker at the generic step; the new model
(v2) is stronger at it. Neither knows the local facts until it reads memory.


<details><summary>Source: <code>ex11_model_swap_context_portability.py</code></summary>

```python
"""Example 11 - Swapping the model: context is the durable asset.

Talk: a newer, smarter model still knows nothing about YOUR environment, because
that knowledge is orthogonal to model intelligence. If your context lives outside the
model (memory files, skills, CLAUDE.md), a new model plugs into it and is instantly
useful; without it, it re-learns every local fact the hard way.

Toy model: 4 local facts must be right (registry, maven, sonar, runner) plus 1
generic hard step. The old model (v1) is weaker at the generic step; the new model
(v2) is stronger at it. Neither knows the local facts until it reads memory.
"""
import tempfile
from memlib import MemoryStore

TRUTH = {"registry": "ai-nexus:5001", "maven": "nexus-mirror", "sonar": "gate-required", "runner": "docker-prod"}
GUESS = {"registry": "docker.io", "maven": "maven-central", "sonar": "optional", "runner": "shared-default"}
MEM = "shared/platform-lessons.md"                # context lives OUTSIDE any model


def run(model, generic_ok, store, session):
    actor = f"agent:{model}"
    original, h = store.read(MEM, actor, missing_ok=True)
    text, failures = original, []
    for topic in TRUTH:
        hits = store.grep(rf"^- {topic}:", "shared")                 # agent greps its memory
        value = hits[0][2].split(": ", 1)[1] if hits else GUESS[topic]
        if value != TRUTH[topic]:
            failures.append(f"{topic}: guessed {value!r}, real value {TRUTH[topic]!r}")
            text += f"- {topic}: {TRUTH[topic]}\n"                    # learns from the error
    if not generic_ok:
        failures.append("generic: cache conflict in multi-stage build (weaker model needs a retry)")
    if text != original:
        store.write(MEM, text, actor, session, expected_hash=h, reason="lessons from failed attempts")
    return failures


def report(label, failures):
    print(f"{label}\n  failed attempts before success: {len(failures)}")
    for f in failures:
        print(f"    - {f}")
    return len(failures)


old_env = MemoryStore(tempfile.mkdtemp())
print("== OLD model (v1) living in this environment ==")
n1 = report("v1, session 1 (empty memory)", run("model-v1", False, old_env, "v1-s1"))
n2 = report("v1, session 2 (memory built by session 1)", run("model-v1", False, old_env, "v1-s2"))

print("\n== NEW smarter model (v2) arrives ==")
n3 = report("v2 COLD: new model, no context attached", run("model-v2", True, MemoryStore(tempfile.mkdtemp()), "v2-cold"))
n4 = report("v2 + SAME context layer (memory from the old model's sessions)", run("model-v2", True, old_env, "v2-ctx"))

print("\nsummary (failed attempts before success): "
      f"v1 first session={n1}, v1 learned={n2}, v2 cold={n3}, v2 + existing context={n4}")
print("-> the model changed; the context layer did not. Intelligence x context.")
```
